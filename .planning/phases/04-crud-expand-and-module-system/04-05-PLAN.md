    ```typescript
    import { Body, Param } from '@nestjs/common'
    import {
      ODataController,
      ODataGet,
      ODataGetByKey,
      ODataPost,
      ODataPatch,
      ODataDelete,
      ODataQueryParam,
      type ODataQuery,
    } from '@nestjs-odata/core'
    import { TypeOrmAutoHandler } from '@nestjs-odata/typeorm'
    import { Request } from 'express'
    import { Req } from '@nestjs/common'

    @ODataController('Products')
    export class ProductsController {
      constructor(private readonly handler: TypeOrmAutoHandler) {}

      @ODataGet('Products')
      async getProducts(
        @ODataQueryParam('Products') query: ODataQuery,
        @Req() req: Request,
      ) {
        return this.handler.handleGet(query, req.originalUrl)
      }

      @ODataGetByKey('Products')
      async getProduct(@Param('key') key: string) {
        return this.handler.handleGetByKey(key, 'Products')
      }

      @ODataPost('Products')
      async createProduct(@Body() body: Record<string, unknown>) {
        return this.handler.handleCreate(body, 'Products')
      }

      @ODataPatch('Products')
      async updateProduct(
        @Param('key') key: string,
        @Body() body: Record<string, unknown>,
      ) {
        return this.handler.handleUpdate(key, body, 'Products')
      }

      @ODataDelete('Products')
      async deleteProduct(@Param('key') key: string) {
        return this.handler.handleDelete(key, 'Products')
      }
    }
    ```

    Note: Since @ODataController already applies the 'Products' prefix at the class level, the method decorators may need path '' (empty) instead of repeating 'Products'. Check how the route resolves — the class-level Controller('Products') combined with method-level Get('') or Get(':key') should give `/Products` and `/Products/:key`. Adjust paths accordingly based on the test results.

    Follow the pattern from existing e2e specs (odata-query.e2e-spec.ts). Set up NestJS testing module with INestApplication.

    Tests (per CRUD-01, CRUD-02, CRUD-03, CRUD-04):

    ```typescript
    describe('OData CRUD Operations', () => {
      // POST /odata/Products
      it('POST creates entity and returns 201 + Location + body', async () => {
        const res = await request(app.getHttpServer())
          .post('/odata/Products')
          .send({ Name: 'Widget', Price: 9.99 })
          .expect(201)

        expect(res.body).toHaveProperty('@odata.context')
        expect(res.body).toHaveProperty('Name', 'Widget')
        expect(res.headers).toHaveProperty('location')
        expect(res.headers.location).toMatch(/Products\(\d+\)/)
      })

      // GET /odata/Products/:key
      it('GET by key returns single entity without value wrapper', async () => {
        // First create, then get by key
        const created = await request(app.getHttpServer())
          .post('/odata/Products')
          .send({ Name: 'Widget', Price: 9.99 })

        const keyId = created.body.Id // or whatever the PK field is

        const res = await request(app.getHttpServer())
          .get(`/odata/Products/${keyId}`)
          .expect(200)

        expect(res.body).toHaveProperty('@odata.context')
        expect(res.body['@odata.context']).toContain('$entity')
        expect(res.body).toHaveProperty('Name', 'Widget')
        expect(res.body).not.toHaveProperty('value') // NOT a collection
      })

      // GET by key 404
      it('GET by key returns 404 for non-existent entity', async () => {
        await request(app.getHttpServer())
          .get('/odata/Products/999999')
          .expect(404)
      })

      // PATCH /odata/Products/:key
      it('PATCH updates entity with merge-patch semantics', async () => {
        const created = await request(app.getHttpServer())
          .post('/odata/Products')
          .send({ Name: 'Widget', Price: 9.99 })

        const keyId = created.body.Id

        const res = await request(app.getHttpServer())
          .patch(`/odata/Products/${keyId}`)
          .send({ Price: 14.99 })
          .expect(200)

        expect(res.body).toHaveProperty('Name', 'Widget')  // unchanged
        expect(res.body).toHaveProperty('Price', 14.99)     // updated
      })

      // PATCH 404
      it('PATCH returns 404 for non-existent entity', async () => {
        await request(app.getHttpServer())
          .patch('/odata/Products/999999')
          .send({ Price: 14.99 })
          .expect(404)
      })

      // DELETE /odata/Products/:key
      it('DELETE removes entity and returns 204', async () => {
        const created = await request(app.getHttpServer())
          .post('/odata/Products')
          .send({ Name: 'Disposable', Price: 1.00 })

        const keyId = created.body.Id

        await request(app.getHttpServer())
          .delete(`/odata/Products/${keyId}`)
          .expect(204)

        // Verify deletion
        await request(app.getHttpServer())
          .get(`/odata/Products/${keyId}`)
          .expect(404)
      })

      // DELETE 404
      it('DELETE returns 404 for non-existent entity', async () => {
        await request(app.getHttpServer())
          .delete('/odata/Products/999999')
          .expect(404)
      })
    })
    ```
