   ```yaml
   name: CI
   on:
     push:
       branches: [main]
     pull_request:
       branches: [main]
   concurrency:
     group: ${{ github.workflow }}-${{ github.ref }}
     cancel-in-progress: true
   jobs:
     lint-and-test:
       runs-on: ubuntu-latest
       strategy:
         matrix:
           node-version: [20, 22]
       steps:
         - uses: actions/checkout@v4
         - uses: pnpm/action-setup@v4
           with:
             version: 10
         - uses: actions/setup-node@v4
           with:
             node-version: ${{ matrix.node-version }}
             cache: 'pnpm'
         - run: pnpm install --frozen-lockfile
         - run: pnpm turbo lint
         - run: pnpm turbo typecheck
         - run: pnpm turbo test
         - run: pnpm turbo build
         - name: Validate package exports (core)
           run: |
             pnpm dlx @arethetypeswrong/cli --pack packages/core
             pnpm dlx publint packages/core
         - name: Validate package exports (typeorm)
           run: |
             pnpm dlx @arethetypeswrong/cli --pack packages/typeorm
             pnpm dlx publint packages/typeorm
   ```
   Per D-13: `@arethetypeswrong/cli` and `publint` run in CI to validate package exports. Uses `--pack` flag which packs the package first to simulate real publish.

2. Create `.github/workflows/release.yml` (per D-10, D-12):
   ```yaml
   name: Release
   on:
     push:
       branches: [main]
   concurrency: ${{ github.workflow }}-${{ github.ref }}
   jobs:
     release:
       runs-on: ubuntu-latest
       permissions:
         contents: write
         pull-requests: write
         id-token: write
       steps:
         - uses: actions/checkout@v4
           with:
             fetch-depth: 0
         - uses: pnpm/action-setup@v4
           with:
             version: 10
         - uses: actions/setup-node@v4
           with:
             node-version: 22
             cache: 'pnpm'
             registry-url: 'https://registry.npmjs.org'
         - run: pnpm install --frozen-lockfile
         - run: pnpm turbo build
         - uses: changesets/action@v1
           with:
             publish: pnpm run release
             title: 'chore: version packages'
             commit: 'chore: version packages'
           env:
             GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
   ```
   Note: No NPM_TOKEN needed. The `id-token: write` permission enables npm OIDC trusted publishing. npm CLI 11.5.1+ auto-detects OIDC environment. The npm Trusted Publisher must be pre-configured on npmjs.com for each package (checkpoint:human-action would be needed at actual publish time, but that is a Phase 5 concern).
