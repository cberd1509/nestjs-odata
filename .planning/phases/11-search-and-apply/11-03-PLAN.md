    ```typescript
    const queryResult = result as ODataQueryResult
    if (queryResult.isAggregated) {
      // Build context URL with aggregated property names as a synthetic SelectNode
      const projectedSelect: SelectNode | undefined = queryResult.applyProperties?.length
        ? { items: queryResult.applyProperties.map((p) => ({ path: [p] })) }
        : undefined
      const contextUrl = buildContextUrl(
        this.options.serviceRoot,
        entitySetName,
        projectedSelect,
      )
      // Aggregated rows: no @odata.id, @odata.type, @odata.navigationLink — just raw data
      const response: Record<string, unknown> = {
        '@odata.context': contextUrl,
        value: queryResult.items,
      }
      if (queryResult.count !== undefined) {
        response['@odata.count'] = queryResult.count
      }
      return response
    }
    ```

| -------------------------------------------- | ----------------------------------------------------------------------- |
| HTTP query string -> ODataQueryPipe          | $search and $apply are untrusted user input parsed in the pipe          |
| ODataQueryResult -> ODataResponseInterceptor | isAggregated flag controls response shape — must not leak internal data |

## STRIDE Threat Register

| Threat ID | Category        | Component                     | Disposition | Mitigation Plan                                                                                                                                 |
| --------- | --------------- | ----------------------------- | ----------- | ----------------------------------------------------------------------------------------------------------------------------------------------- |
| T-11-08   | Tampering       | odata-query.pipe.ts           | mitigate    | $search and $apply values are URL-decoded then parsed by validated parsers (Plan 01); invalid input throws ODataParseError -> 400               |
| T-11-09   | Info Disclosure | odata-response.interceptor.ts | mitigate    | Aggregated response uses explicit applyProperties to build context URL; raw SQL column names not exposed if apply-visitor uses explicit aliases |
| T-11-10   | Tampering       | odata-query.pipe.ts           | accept      | $apply field names are not validated against EDM in the pipe (validation happens in translator); this is by design per RESEARCH recommendation  |
