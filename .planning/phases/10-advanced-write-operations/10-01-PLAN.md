```typescript
import { applyDecorators, Patch, SetMetadata, UseFilters, UseInterceptors } from '@nestjs/common'
import { ODATA_ROUTE_KEY } from './metadata-keys.js'
import { ODataResponseInterceptor } from '../response/odata-response.interceptor.js'
import { ODataExceptionFilter } from '../response/odata-exception.filter.js'

export interface ODataPatchOptions {
  path?: string
}

export function ODataPatch(entitySetName: string, options?: ODataPatchOptions): MethodDecorator {
  return applyDecorators(
    Patch(options?.path ?? ':key'),
    SetMetadata(ODATA_ROUTE_KEY, {
      entitySetName,
      operation: 'update',
      isSingleEntity: true,
    }),
    UseInterceptors(ODataResponseInterceptor),
    UseFilters(ODataExceptionFilter),
  )
}
```

## Trust Boundaries

| Boundary               | Description                                                                                       |
| ---------------------- | ------------------------------------------------------------------------------------------------- |
| client -> PUT endpoint | Untrusted body may contain extra fields, mismatched keys, or attempt to overwrite managed columns |
| client -> $batch PUT   | Same risks as direct PUT, plus Content-ID injection (addressed in Plan 02)                        |

## STRIDE Threat Register

| Threat ID | Category  | Component                       | Disposition | Mitigation Plan                                                                                                                         |
| --------- | --------- | ------------------------------- | ----------- | --------------------------------------------------------------------------------------------------------------------------------------- |
| T-10-01   | Tampering | handleReplace body key mismatch | mitigate    | Reject body key != URL key before any DB write (per D-01). Validate via parseODataKey typed comparison.                                 |
| T-10-02   | Tampering | handleReplace mass assignment   | mitigate    | TypeORM repo.create() only maps declared entity columns -- unknown body fields silently ignored. Same pattern as existing handleCreate. |
| T-10-03   | Tampering | handleReplace managed columns   | mitigate    | Skip isPrimary, isCreateDate, isUpdateDate, isVersion columns in replacement construction -- cannot be overwritten via PUT body.        |
| T-10-04   | Elevation | handleReplace bypassing ETag    | mitigate    | If-Match enforcement copied from handleUpdate -- identical ETag validation block. 412 on mismatch.                                      |
