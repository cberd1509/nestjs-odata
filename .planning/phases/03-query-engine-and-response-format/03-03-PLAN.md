    ```typescript
    import type { SelectNode } from '../parser/ast.js'

    export function buildContextUrl(serviceRoot: string, entitySetName: string, select?: SelectNode): string {
      // Normalize: strip trailing slash from serviceRoot
      const root = serviceRoot.endsWith('/') ? serviceRoot.slice(0, -1) : serviceRoot
      const base = `${root}/$metadata#${entitySetName}`
      // Per D-07: if select has specific items (not all, not undefined), append (Field1,Field2)
      if (select?.items?.length) {
        const fields = select.items.map(item => item.path.join('/')).join(',')
        return `${base}(${fields})`
      }
      return base
    }
    ```

    import { applyDecorators, Get, SetMetadata, UseFilters, UseInterceptors } from '@nestjs/common'
    import { ODATA_ROUTE_KEY } from './metadata-keys.js'
    import { ODataResponseInterceptor } from '../response/odata-response.interceptor.js'

    export interface ODataGetOptions {
      path?: string
      autoHandler?: boolean
    }

    export function ODataGet(entitySetName: string, options?: ODataGetOptions): MethodDecorator {
      return applyDecorators(
        Get(options?.path ?? ''),
        SetMetadata(ODATA_ROUTE_KEY, { entitySetName, autoHandler: options?.autoHandler ?? false }),
        UseInterceptors(ODataResponseInterceptor),
        UseFilters(ODataExceptionFilter),
      )
    }

<threat_model>

## Trust Boundaries

| Boundary                         | Description                                                            |
| -------------------------------- | ---------------------------------------------------------------------- |
| Handler result -> HTTP response  | ODataResponseInterceptor transforms internal result to external JSON   |
| Exception -> HTTP error response | ODataExceptionFilter transforms internal errors to external error body |

## STRIDE Threat Register

| Threat ID | Category        | Component                | Disposition | Mitigation Plan                                                                                                                                                                                                                                     |
| --------- | --------------- | ------------------------ | ----------- | --------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------- |
| T-03-08   | Info Disclosure | ODataExceptionFilter     | mitigate    | Generic errors return "An unexpected error occurred." — never exception.stack, never internal file paths, never SQL details. Only ODataParseError and ODataValidationError expose their message (which is user-input-derived, not server-internal). |
| T-03-09   | Spoofing        | ODataResponseInterceptor | mitigate    | Interceptor only activates on routes with ODATA_ROUTE_KEY metadata. Cannot be triggered on non-OData routes. Metadata is set by @ODataGet() at decoration time, not at request time.                                                                |
| T-03-10   | Info Disclosure | buildContextUrl          | accept      | Context URL reveals serviceRoot path and entity set name. This is intentional per OData v4 spec and is the same information the client already knows from the request URL.                                                                          |

</threat_model>
